% Note: use serialportlist() to check for available serial ports
serialPortName = "COM3";

% define packet specifier
packetSpec = {'2*uint8', '2*int32', '2*single', 'single'};

% select baud rate
baudRate = 38400;

% configure scopes
plotsGridSize = [4, 1];

    1, 1, 2; ... %
    2, 2, 1; ... %
    2, 2, 2; ... %
    3, 3, 1; ... %
    3, 3, 2; ... %
    4, 4, 1];    %

% start serialDataLoggerApp with specified input arguments
serialDataLoggerApp( ...
    "SerialPortName", serialPortName, ...
    % R. Antonello, F. Ticozzi Control Engineering Laboratory 14/16
    "PacketSpec", packetSpec, ...
    "BaudRate", baudRate, ...
    "EnableStartStopCmds", true, ...
    "TxStartCmd", 65, ...
    "TxStopCmd", 66, ...
    "EnableCOBS", true, ...
    "EnableNullTerminator", true, ...
    "BufferSize", 10000, ...
    "TxSampleTime", 0.01, ...
    "EnableLogToWorkspace", true, ...
    "EnableSaveOnExit", true, ...
    "VarName", "data", ...
    "PlotsGridSize", plotsGridSize, ...
    "PlotsContentList", plotsContentList, ...
    "PlotsDuration", 5, ...
    "PlotsRefreshRatio", 8);
